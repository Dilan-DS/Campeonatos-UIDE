# Dependencias directas del proyecto, con version fija.
#
# Se fijan a proposito: sin version, un `pip install -r` en otra fecha o en
# otra maquina puede traer versiones distintas y romper el proyecto sin que
# nadie haya tocado el codigo. Para subir una version, cambiala aqui y
# comprueba que CI sigue verde.

Django==5.2.17

# Lee la configuracion desde .env (CampeonatosUIDE/settings.py).
django-environ==0.14.0

# Clases de Bulma en los widgets de los formularios ({% render_field %}).
django-widget-tweaks==1.5.1

# MultiSelectField en los modelos de deportes.
django-multiselectfield==1.0.1

# Exportaciones a Excel (campeonato_views, estadisticas_views).
openpyxl==3.1.5
pandas==3.0.5

# Exportaciones a PDF.
reportlab==5.0.1

# Requerida por los ImageField de los modelos (logos, comprobantes,
# noticias, galeria). Hasta ahora solo entraba de rebote como dependencia
# de reportlab; se declara porque el proyecto la usa de forma directa.
pillow==12.3.0

# WSGI server used by the production container.
gunicorn==23.0.0

# Sirve los estaticos (incluido el admin) desde el propio proceso Django en
# produccion. Sin esto, con DEBUG=False Django no entrega /static/ y Nginx,
# que solo hace proxy, devuelve 404 en todo el CSS/JS.
whitenoise==6.8.2

# Driver MySQL en Python puro para el contenedor de produccion. Ver
# CampeonatosUIDE/__init__.py. En local con SQLite no se usa.
# El extra [rsa] arrastra "cryptography", que PyMySQL necesita para el
# metodo de autenticacion por defecto de MySQL 8 (caching_sha2_password).
PyMySQL[rsa]==1.1.1
